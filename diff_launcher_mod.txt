+fn get_primary_work_area() -> (i32, i32, i32, i32) {
+    use windows::Win32::{
+        Foundation::POINT,
+        Graphics::Gdi::{GetMonitorInfoW, MonitorFromPoint, MONITORINFO, MONITOR_DEFAULTTOPRIMARY},
+    };
+
+    let monitor = unsafe {
+        MonitorFromPoint(
+            POINT { x: 0, y: 0 },
+            MONITOR_DEFAULTTOPRIMARY,
+        )
+    };
+    if monitor.0.is_null() {
+        return fallback();
+    }
+
+    let mut monitor_info = MONITORINFO {
+        cbSize: std::mem::size_of::<MONITORINFO>() as u32,
+        ..Default::default()
+    };
+    let succeeded = unsafe { GetMonitorInfoW(monitor, &mut monitor_info).as_bool() };
+    if !succeeded {
+        return fallback();
+    }
+
+    let rect = monitor_info.rcWork;
+    (rect.left, rect.top, rect.right - rect.left, rect.bottom - rect.top)

+#[cfg(target_os = "windows")]
+fn set_launcher_window_geometry(window: &Window, left: i32, top: i32, width: i32, height: i32) {
+    use raw_window_handle::{HasWindowHandle, RawWindowHandle};
+    use windows::Win32::{
+        Foundation::{HWND, RECT},
+        UI::WindowsAndMessaging::{
+            SetWindowPos, SWP_NOACTIVATE, SWP_NOREDRAW, SWP_NOSIZE, SWP_NOZORDER, GetWindowRect,
+        },
+    };
+
+    if let Ok(window_handle) = HasWindowHandle::window_handle(window) {
+        if let RawWindowHandle::Win32(win32_window_handle) = window_handle.as_raw() {
+            unsafe {
+                let hwnd = HWND(win32_window_handle.hwnd.get() as *mut std::ffi::c_void);
+                let mut rect = RECT::default();
+                let _ = GetWindowRect(hwnd, &mut rect);
+                let current_width = rect.right - rect.left;
+                let current_height = rect.bottom - rect.top;
+
+                let mut flags = SWP_NOZORDER | SWP_NOACTIVATE;
+                if current_width == width && current_height == height {
+                    // Size unchanged — skip resize and suppress Windows-side repaint.
+                    // GPUI continues to render via its own compositor, so we only
+                    // need the OS to move the surface without touching its content.
+                    flags |= SWP_NOSIZE | SWP_NOREDRAW;
+                }
+
+                let _ = SetWindowPos(
+                    hwnd,
+                    None,
+                    left,
+                    top,
+                    width,
+                    height,
+                    flags,
+                );
+            }
+        }
+    }
+}
